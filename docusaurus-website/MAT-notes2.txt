```python
import datetime

def get_current_timestamp() -> str:
    """دریافت تاریخ و زمان فعلی به صورت فرمت شده"""
    now = datetime.datetime.now()
    
    # فرمت فارسی
    jalali_date = _convert_to_jalali(now)
    
    return f"{jalali_date} - {now.strftime('%H:%M')}"

def _convert_to_jalali(gregorian_date: datetime.datetime) -> str:
    """تبدیل تاریخ میلادی به شمسی (ساده شده)"""
    # در صورت نصب بودن jdatetime:
    try:
        import jdatetime
        jalali = jdatetime.datetime.fromgregorian(datetime=gregorian_date)
        return jalali.strftime("%Y/%m/%d")
    except ImportError:
        # اگر jdatetime نصب نیست، از تاریخ میلادی استفاده کن
        return gregorian_date.strftime("%Y/%m/%d")
```

```python
def create_filename(en_title: str, fa_title: str = None) -> str:
    """ایجاد نام فایل انگلیسی از عنوان انگلیسی یا فارسی"""
    # اولویت با عنوان انگلیسی
    if en_title and en_title != 'nan' and not pd.isna(en_title):
        filename = _slugify_en(str(en_title))
    elif fa_title and fa_title != 'nan' and not pd.isna(fa_title):
        filename = _slugify_fa(str(fa_title))
    else:
        return ''
    
    return filename

def _slugify_en(text: str) -> str:
    """ایجاد اسلاگ از متن انگلیسی"""
    if not text:
        return ''
    
    # تبدیل به حروف کوچک
    text = text.lower()
    
    # جایگزینی کاراکترهای خاص
    replacements = {
        '&': 'and',
        '@': 'at',
        '%': 'percent',
        '+': 'plus',
        '#': 'sharp',
        '?': '',
        '!': '',
        ':': '',
        ';': '',
        "'": '',
        '"': '',
        '`': '',
        '~': '',
        '(': '',
        ')': '',
        '[': '',
        ']': '',
        '{': '',
        '}': '',
        '<': '',
        '>': '',
        '/': '-',
        '\\': '-',
        '|': '-',
        '=': '-',
    }
    
    for old, new in replacements.items():
        text = text.replace(old, new)
    
    # حذف کاراکترهای غیرمجاز
    text = re.sub(r'[^\w\s\-\.]', '', text)
    
    # جایگزینی فاصله با خط تیره
    text = re.sub(r'[\s_]+', '-', text)
    
    # حذف خط تیره‌های تکراری
    text = re.sub(r'\-+', '-', text)
    
    # حذف خط تیره از ابتدا و انتها
    text = text.strip('-')
    
    return text

def _slugify_fa(text: str) -> str:
    """ایجاد اسلاگ از متن فارسی (ترجمه ساده)"""
    if not text:
        return ''
    
    # نگاشت ساده فارسی به انگلیسی برای کلمات رایج
    persian_to_english = {
        'ریاضی': 'mathematics',
        'عمومی': 'general',
        'آمار': 'statistics',
        'احتمال': 'probability',
        'مهندسی': 'engineering',
        'برنامه‌نویسی': 'programming',
        'داده': 'data',
        'ساختار': 'structure',
        'پایگاه': 'database',
        'یادگیری': 'learning',
        'ماشین': 'machine',
        'هوش': 'intelligence',
        'محاسبات': 'computing',
        'الگوریتم': 'algorithm',
        'کامپیوتر': 'computer',
        'شبکه': 'network',
        'نرم‌افزار': 'software',
        'سخت‌افزار': 'hardware',
        'پروژه': 'project',
        'کارگاه': 'workshop',
        'زبان': 'language',
        'انگلیسی': 'english',
        'فارسی': 'persian',
        'تخصصی': 'specialized',
        'مهارتی': 'skill',
        'پایه': 'basic',
        'الزامی': 'mandatory',
        'اختیاری': 'elective',
        'تحقیق': 'research',
        'روش': 'method',
        'علمی': 'scientific',
        'کاربردی': 'applied'
    }
    
    # جایگزینی کلمات فارسی با معادل انگلیسی
    for persian, english in persian_to_english.items():
        text = text.replace(persian, english)
    
    # حذف کاراکترهای فارسی باقی‌مانده
    text = re.sub(r'[^\x00-\x7F]+', '-', text)
    
    # پاکسازی نهایی
    text = _slugify_en(text)
    
    return text
```

```python
def _create_symbolic_links():
    """ایجاد symbolic link از category/ به curriculum/"""
    import os
    
    categories = ['base', 'mandatory', 'elective', 'skill', 'other']
    
    # ایجاد پوشه category اگر وجود ندارد
    os.makedirs('./docs/category', exist_ok=True)
    
    for category in categories:
        src = f'../curriculum/{category}'
        dst = f'./docs/category/{category}'
        
        # اطمینان از وجود پوشه مبدا
        os.makedirs(f'./docs/curriculum/{category}', exist_ok=True)
        
        # اگر symbolic link قبلی وجود دارد، حذف کن
        if os.path.exists(dst):
            if os.path.islink(dst):
                os.unlink(dst)
            elif os.path.isdir(dst):
                import shutil
                shutil.rmtree(dst)
        
        # ایجاد symbolic link
        try:
            if os.name == 'nt':  # ویندوز
                import _winapi
                _winapi.CreateJunction(os.path.abspath(src), dst)
                print(f"✅ Created junction (Windows): {dst} -> {src}")
            else:  # Linux/Mac
                os.symlink(os.path.abspath(src), dst, target_is_directory=True)
                print(f"✅ Created symlink: {dst} -> {src}")
        except:
            # اگر symbolic link نشد، پوشه را کپی کن
            import shutil
            if os.path.exists(src):
                shutil.copytree(src, dst, dirs_exist_ok=True)
                print(f"✅ Copied directory: {src} -> {dst}")
```
